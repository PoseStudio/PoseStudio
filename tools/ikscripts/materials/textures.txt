# THE TEXTURE SLOTS (2026-11-09): the Materials tab's Textures group driven on the base figure —
#   POSESTUDIO_TEST_FIGURE=<figure> tools/ikscripts/run.sh tools/ikscripts/materials/textures.txt <out> POSESTUDIO_IK_SCRIPT_ASSETS=<assets>
# (the PNGs from tools/materials/make_test_assets.py; a relative path in `texture` resolves
# against POSESTUDIO_IK_SCRIPT_ASSETS, else the output folder). The torso imports with a
# base-colour map and no cutout: a file replaces the map, undo brings the import's back, the slot
# is cleared and restored; a cutout chosen on a zone without one makes it a cutout zone and the
# arrow takes it away again; a project round trip carries a chosen map and a cleared slot.
view front
click chest
expect zonename.Torso == 1
expect matmap.basecolor == 1
expect mattex.basecolor == 1
texture basecolor tex_red.png
expect mattex.basecolor == 2
expect matmap.basecolor == 1
closeup chest 0.6
shot 01_red_basecolor
undo
expect mattex.basecolor == 1
texture basecolor none
expect mattex.basecolor == 0
expect matmap.basecolor == 0
shot 02_no_basecolor
texture basecolor imported
expect mattex.basecolor == 1
expect matmap.basecolor == 1
expect matmap.cutout == 0
texture cutout tex_mask.png
expect matmap.cutout == 1
expect mattex.cutout == 2
shot 03_cutout_checker
texture cutout imported
expect matmap.cutout == 0
texture roughness tex_rough.png
expect mattex.roughness == 2
expect matmap.roughness == 1
shot 04_roughness_ramp
texture roughness imported
expect mattex.roughness < 2
texture emission tex_red.png
expect matmap.emission == 1
texture emission none
expect matmap.emission == 0
# The project round trip: a chosen base colour and a cleared normal slot come back.
texture basecolor tex_red.png
texture normal none
saveproject textures.pss
new
expect figures == 0
openproject textures.pss
click chest
expect zonename.Torso == 1
expect mattex.basecolor == 2
expect mattex.normal == 0
shot 05_after_project
texture basecolor imported
expect mattex.basecolor == 1
